# ===== 1. 准备环境 =====
cd ai-chat-api

# 创建 .env 文件（填入你的 API Key）
cp .env.example .env
# 编辑 .env: 填 DEEPSEEK_API_KEY 和（可选）OPENAI_API_KEY

# 创建虚拟环境
python3 -m venv .venv
source .venv/bin/activate  # macOS/Linux
# .venv\Scripts\activate    # Windows

# 安装依赖
pip install -r requirements.txt


# ===== 2. 本地运行 =====
uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload

# 打开浏览器: http://localhost:8000/docs
# 你会看到自动生成的 Swagger 文档界面


# ===== 3. 测试接口 =====
# 测试 /health
curl http://localhost:8000/health
# → {"status":"healthy","version":"0.1.0",...}

# 查看可用模型
curl http://localhost:8000/models
# → {"models":[{"id":"deepseek","name":"DeepSeek-V3",...}],...}

# 发送聊天请求
curl -X POST http://localhost:8000/chat \
  -H "Content-Type: application/json" \
  -d '{
    "model": "deepseek",
    "messages": [
      {"role": "system", "content": "你是 Python 编程助手"},
      {"role": "user", "content": "用一句话介绍 Python"}
    ],
    "temperature": 0.7
  }'
# → {"id":"chat-abc123","model":"deepseek-chat","content":"Python 是...","usage":{...}}

# 流式输出（在终端中逐 Token 显示）
curl -X POST http://localhost:8000/chat/stream \
  -H "Content-Type: application/json" \
  -d '{
    "model": "deepseek",
    "messages": [{"role": "user", "content": "讲一个简短的笑话"}],
    "temperature": 0.7
  }'


# ===== 4. 运行测试 =====
pytest tests/ -v
# → tests/test_health.py::TestHealth::test_liveness PASSED
# → tests/test_health.py::TestHealth::test_readiness PASSED
# → tests/test_chat.py::TestChat::test_chat_returns_response PASSED
# → ... 全部通过！


# ===== 5. Docker 部署 =====
docker compose up -d --build
# → 构建镜像 → 启动容器

# 验证 Docker 部署
curl http://localhost:8000/health
# → {"status":"healthy",...}

# 查看日志
docker compose logs -f api

# 停止
docker compose down
